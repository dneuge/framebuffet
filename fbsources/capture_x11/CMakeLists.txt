CMAKE_MINIMUM_REQUIRED(VERSION 3.22.1)
PROJECT(fbsource_capture_x11)

SET(CMAKE_BUILD_TYPE Debug)
SET(CMAKE_POSITION_INDEPENDENT_CODE TRUE)

FIND_PACKAGE(X11 REQUIRED)
INCLUDE_DIRECTORIES(${X11_INCLUDE_PATH})

FIND_PACKAGE(PkgConfig REQUIRED)
PKG_CHECK_MODULES(xcomposite REQUIRED xcomposite)
INCLUDE_DIRECTORIES(${XCOMPOSITE_INCLUDE_DIRS})
#PKG_CHECK_MODULES(xext REQUIRED xext)
#INCLUDE_DIRECTORIES(${XEXT_INCLUDE_DIRS})

SET(FRAMEBUFFET_INCLUDE ${CMAKE_CURRENT_SOURCE_DIR}/../../framework/release/include)
SET(FRAMEBUFFET_LIB ${CMAKE_CURRENT_SOURCE_DIR}/../../framework/release/lib)

SET(FBPLUGIN_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/src/capture_x11.c
)

SET(LIB_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/../../lib/_build")

ADD_DEFINITIONS(-DFBLOG_COMPILED_MIN_LOG_LEVEL=FBLOG_LEVEL_TRACE)

ADD_LIBRARY(fbsource_capture_x11 SHARED ${FBPLUGIN_SOURCES})

ADD_LIBRARY(fbsource_capture_x11_static STATIC ${FBPLUGIN_SOURCES})
TARGET_COMPILE_DEFINITIONS(fbsource_capture_x11_static PRIVATE FBSOURCE_BUILD_STATIC_MODULE)

TARGET_INCLUDE_DIRECTORIES(fbsource_capture_x11 PUBLIC ${FRAMEBUFFET_INCLUDE} ${X11_X11_INCLUDE_PATH} ${XCB_INCLUDE_DIRS})
TARGET_INCLUDE_DIRECTORIES(fbsource_capture_x11_static PUBLIC ${FRAMEBUFFET_INCLUDE} ${X11_X11_INCLUDE_PATH} ${XCB_INCLUDE_DIRS})

TARGET_LINK_LIBRARIES(fbsource_capture_x11 ${X11_LIBRARIES} Xcomposite Xext)

# compiled output file name
SET_TARGET_PROPERTIES(fbsource_capture_x11 PROPERTIES PREFIX "")
SET_TARGET_PROPERTIES(fbsource_capture_x11 PROPERTIES OUTPUT_NAME "fbsource_capture_x11")

SET_TARGET_PROPERTIES(fbsource_capture_x11_static PROPERTIES PREFIX "")
SET_TARGET_PROPERTIES(fbsource_capture_x11_static PROPERTIES OUTPUT_NAME "fbsource_capture_x11")
