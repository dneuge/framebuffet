CMAKE_MINIMUM_REQUIRED(VERSION 3.22.1)
PROJECT(fbsource_merge)

INCLUDE(CheckIncludeFile)

SET(CMAKE_BUILD_TYPE Debug)

SET(FRAMEBUFFET_INCLUDE ${CMAKE_CURRENT_SOURCE_DIR}/../../framework/release/include)
SET(FRAMEBUFFET_LIB ${CMAKE_CURRENT_SOURCE_DIR}/../../framework/release/lib)

SET(FBPLUGIN_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/src/merge.c
)

SET(LIB_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/../../lib/_build")

ADD_DEFINITIONS(-DFBLOG_COMPILED_MIN_LOG_LEVEL=FBLOG_LEVEL_TRACE)

# check if C11 compatibility wrappers are needed for threads.h and subsequently time.h
CHECK_INCLUDE_FILE(threads.h HAVE_C11_THREADS)
IF(HAVE_C11_THREADS)
    ADD_DEFINITIONS(-DHAVE_C11_THREADS)
ELSE()
    SET(LINK_C11_COMPAT_STATIC "${LIB_BUILD}/libmesa_c11.a")
ENDIF()

SET(FBPLUGIN_LINK_OPTIONS "")
#SET(FBPLUGIN_LINK_OPTIONS -Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/export_symbols.map)

SET(LINK_FRAMEBUFFET_DYNAMIC "")

IF(UNIX)
    ADD_LIBRARY(framebuffet INTERFACE)
    
    ADD_LIBRARY(fbsource_merge SHARED ${FBPLUGIN_SOURCES})
ELSEIF(WIN32)
    SET(LINK_FRAMEBUFFET_DYNAMIC "${FRAMEBUFFET_LIB}/libframebuffet_dll_shim.a")

    ADD_LIBRARY(fbsource_merge MODULE ${FBPLUGIN_SOURCES})

    # libgcc_s_seh will be missing on Windows if we don't compile with static-libgcc
    TARGET_LINK_OPTIONS(fbsource_merge PRIVATE -static-libgcc ${FBPLUGIN_LINK_OPTIONS})
ENDIF()

ADD_LIBRARY(fbsource_merge_static STATIC ${FBPLUGIN_SOURCES})
TARGET_COMPILE_DEFINITIONS(fbsource_merge_static PRIVATE FBSOURCE_BUILD_STATIC_MODULE)

TARGET_INCLUDE_DIRECTORIES(fbsource_merge PUBLIC ${FRAMEBUFFET_INCLUDE} ${LIB_BUILD})
TARGET_INCLUDE_DIRECTORIES(fbsource_merge_static PUBLIC ${FRAMEBUFFET_INCLUDE} ${LIB_BUILD})

TARGET_LINK_LIBRARIES(fbsource_merge ${LINK_FRAMEBUFFET_DYNAMIC} ${LINK_C11_COMPAT_STATIC})
TARGET_LINK_LIBRARIES(fbsource_merge_static ${LINK_C11_COMPAT_STATIC})

# compiled output file name
SET_TARGET_PROPERTIES(fbsource_merge PROPERTIES PREFIX "")
SET_TARGET_PROPERTIES(fbsource_merge PROPERTIES OUTPUT_NAME "fbsource_merge")

SET_TARGET_PROPERTIES(fbsource_merge_static PROPERTIES PREFIX "")
SET_TARGET_PROPERTIES(fbsource_merge_static PROPERTIES OUTPUT_NAME "fbsource_merge")
